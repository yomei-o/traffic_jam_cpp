cmake_minimum_required(VERSION 3.16)
project(traffic_jam C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  # mediakit と同じ静的 CRT（VC 再頒布パッケージ無しで動く）
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options(/utf-8)
endif()

# mp4 の展開・書き出しは mediakit_cpp。隣に clone してあればそれを、無ければ GitHub から取ってくる。
set(MEDIAKIT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../mediakit_cpp" CACHE PATH "mediakit_cpp のソース")
set(MK_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(MK_BUILD_TESTS OFF CACHE BOOL "" FORCE)
if(EXISTS "${MEDIAKIT_DIR}/CMakeLists.txt")
  add_subdirectory("${MEDIAKIT_DIR}" mediakit EXCLUDE_FROM_ALL)
else()
  include(FetchContent)
  FetchContent_Declare(mediakit GIT_REPOSITORY https://github.com/yomei-o/mediakit_cpp.git GIT_TAG main)
  FetchContent_GetProperties(mediakit)
  if(NOT mediakit_POPULATED)
    FetchContent_Populate(mediakit)
    add_subdirectory(${mediakit_SOURCE_DIR} ${mediakit_BINARY_DIR} EXCLUDE_FROM_ALL)
  endif()
endif()

# 車両検出器は vehicle_det_cpp（ヘッダーオンリー）。隣に clone してあればそれを、無ければ GitHub から取ってくる。
set(VEHICLE_DET_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../vehicle_det_cpp" CACHE PATH "vehicle_det_cpp のソース")
if(NOT EXISTS "${VEHICLE_DET_DIR}/vehicle_det.hpp")
  include(FetchContent)
  FetchContent_Declare(vehicle_det GIT_REPOSITORY https://github.com/yomei-o/vehicle_det_cpp.git GIT_TAG main)
  FetchContent_GetProperties(vehicle_det)
  if(NOT vehicle_det_POPULATED)
    FetchContent_Populate(vehicle_det)
  endif()
  set(VEHICLE_DET_DIR ${vehicle_det_SOURCE_DIR})
endif()

add_executable(traffic_jam main.cpp)
target_include_directories(traffic_jam PRIVATE ${VEHICLE_DET_DIR} ${VEHICLE_DET_DIR}/pure ${VEHICLE_DET_DIR}/pure/third_party)
target_link_libraries(traffic_jam PRIVATE mk_codec)
if(MSVC)
  target_compile_definitions(traffic_jam PRIVATE NOMINMAX)
  target_compile_options(traffic_jam PRIVATE /bigobj /openmp /wd4244 /wd4267 /wd4305 /wd4996)
endif()
